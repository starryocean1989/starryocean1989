def _initial_quality_scan(self):
    """初始数据质量扫描（后台线程）

    ⚠️ 重要：必须在品种列表更新完成后才开始扫描
    """
    # 步骤1：等待引擎完全初始化
    time.sleep(3)
    self.logger.info("🔍 准备启动数据质量扫描...")

    # 步骤2：等待品种列表加载/更新完成
    max_wait_time = 60  # 最多等待60秒
    wait_interval = 2   # 每2秒检查一次
    elapsed = 0

    while elapsed < max_wait_time:
        try:
            current_codes = self.symbol_loader.extract_all_codes()

            if len(current_codes) > 0:
                self.logger.info(f"✓ 品种列表已就绪: {len(current_codes)}个品种")
                break
            else:
                self.logger.debug(f"品种列表为空，继续等待... ({elapsed}s/{max_wait_time}s)")
                time.sleep(wait_interval)
                elapsed += wait_interval
        except Exception as e:
            self.logger.warning(f"检查品种列表失败: {e}，继续等待...")
            time.sleep(wait_interval)
            elapsed += wait_interval

    if elapsed >= max_wait_time:
        self.logger.warning("⚠️ 等待品种列表超时，使用当前可用品种开始扫描")

    # 步骤3：开始数据质量扫描
    try:
        final_count = len(self.symbol_loader.extract_all_codes())
        self.logger.info(f"🚀 开始初始数据质量扫描（品种数: {final_count}）...")

        overview = self.trigger_data_quality_scan(force_refresh=True)

        if overview:
            self._push_quality_overview_event(overview)
            self.logger.info(
                f"✓ 初始数据质量扫描完成，品种: {overview.total_symbols}，评分: {overview.quality_score}"
            )
    except Exception as e:
        self.logger.error(f"初始数据质量扫描失败: {e}", exc_info=True)
```

def _load_quality_overview_async(self):
    # 设置状态指示器：正在扫描
    self._set_quality_scan_status(True)

    def load_in_background():
        try:
            result = self.data_center_service.get_data_quality_overview()

            if result.get("success"):
                # 更新UI
                QTimer.singleShot(0, lambda: [
                    self._update_quality_overview_ui(result),
                    self._set_quality_scan_status(False)  # 扫描完成
                ])
        except Exception as e:
            self.logger.error(f"加载失败: {e}")
            self._set_quality_scan_status(False)

    threading.Thread(target=load_in_background, daemon=True).start()

def scan_all_data(self, reference_symbols, intervals=None, force_refresh=False):
    """扫描所有数据质量（优化版：并发扫描，智能过滤）"""

    # 1. 获取本地有数据的品种
    local_data_index = self.storage_manager.get_local_data_index()
    symbols_with_data = [s for s in reference_symbols if s in local_data_index]

    self.logger.info(
        f"开始数据质量扫描: 总品种{len(reference_symbols)}个，"
        f"本地有数据{len(symbols_with_data)}个（将并发扫描）"
    )

    # 2. 并发扫描（10个线程）
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = {executor.submit(scan_single, s): s
                   for s in symbols_with_data}

        # 进度日志
        for future in as_completed(futures):
            symbol_quality = future.result()
            if symbol_quality:
                symbol_qualities.append(symbol_quality)
                completed += 1
                if completed % 500 == 0:
                    self.logger.info(f"扫描进度: {completed}/{len(symbols_with_data)}")

    # 3. 过滤只保留有问题的品种
    problem_details = [
        {
            "symbol": sq.symbol,
            "status": self._determine_status(sq),
            "score": sq.overall_score,
            "issues": self._collect_issues(sq)
        }
        for sq in symbol_qualities
        if sq.has_errors or sq.has_warnings or sq.is_missing
    ]

    # 4. 生成质量概览
    return QualityOverview(
        total_symbols=len(reference_symbols),  # 6135
        details=problem_details  # 只有问题品种
    )
```

#### 3.3 检查缺失日期（核心逻辑）

**关键改进**: 整合基日、上市日期、交易日历、最近交易日四个因素

```python
def _check_missing_dates(self, df: pd.DataFrame, symbol: str) -> List[date]:
    # 1. 提取实际日期
    actual_dates = self._convert_to_date_set(date_series)
    data_start = min(actual_dates)
    data_end = max(actual_dates)

    # 2. 获取基日（默认2020-01-01）
    base_date = config_manager.get_base_date()

    # 3. 获取上市日期（已修复API调用）
    ipo_date = self._get_ipo_date(symbol)

    # 4. 确定有效起点（三者取最大值）
    effective_start = max(data_start, base_date, ipo_date if ipo_date else base_date)

    # 5. 获取最近交易日（不检测未来日期）
    latest_trading_day = self._get_latest_trading_day()

    # 6. 确定检测终点（不超过最近交易日）
    check_end_date = min(data_end, latest_trading_day)

    # 7. 验证日期范围（防止倒置）
    if effective_start > check_end_date:
        self.logger.warning(
            f"品种 {symbol} 日期范围异常: "
            f"effective_start({effective_start}) > check_end_date({check_end_date}), "
            f"跳过缺失检测"
        )
        return []

    # 8. 获取交易日范围（只包含交易日，不含周末节假日）
    expected_trading_days = self._get_trading_days_range(
        effective_start,
        check_end_date
    )

    # 9. 计算缺失日期
    expected_dates = {datetime.strptime(d, "%Y-%m-%d").date()
                      for d in expected_trading_days}
    missing_dates = list(expected_dates - actual_dates)

    return missing_dates
```

**关键因素说明**:

1. **基日过滤** (`base_date`):
   - 默认: 2020-01-01
   - 作用: 基日之前的数据不算缺失
   - 来源: `config_manager.get_base_date()`

2. **上市日过滤** (`ipo_date`):
   - 来源: TDX API (`get_finance_info`)
   - 作用: 上市前的数据不算缺失
   - 修复: 不再使用 `async with`，改为 `try/finally + api.close()`

3. **交易日过滤** (`TradingCalendar`):
   - 来源: `pandas_market_calendars` (XSHG)
   - 作用: 周末和节假日不算缺失
   - 缓存: 24小时TTL

4. **未来日过滤** (`latest_trading_day`):
   - 作用: 只检测到最近交易日，不检测未来
   - 来源: `TradingCalendar.get_previous_trading_day()`

5. **日期范围验证**:
   - 作用: 防止日期倒置（start > end）
   - 触发: 当 `effective_start > check_end_date` 时跳过检测

### 1. 启动顺序优化

### 2. API调用修复

**问题**: `async with api: AttributeError: __aenter__`

**解决**:
```python
# 错误写法
async with api:
    finance_info = await api.get_finance_info(...)

# 正确写法
api = await AsyncTdxHq_API.factory(server)
try:
    finance_info = await api.get_finance_info(...)
finally:
    await api.close()
```

### 3. 缺失日期智能检测

**整合四个因素**:
1. 基日（2020-01-01）- 之前不算缺失

3. 交易日历（pandas_market_calendars）- 周末节假日不算缺失
4. 最近交易日（动态）- 未来日期不算缺失

- 后端过滤，只返回有问题的品种
- 前端直接显示，不再遍历所有品种
- 无问题时显示友好提示

## 技术细节说明

### 为什么要等待品种列表？

1. **数据完整性**
   - 品种列表是数据质量扫描的基础
   - 如果列表不完整，扫描结果就不准确

2. **避免重复扫描**
   - 如果先扫描5392个，后来发现还有743个
   - 需要重新扫描，浪费资源

3. **用户体验**
   - 显示正确的总品种数
   - 显示准确的质量统计

### 为什么最多等待60秒？

1. **防止死锁**
   - 如果品种列表更新失败，不能无限等待
   - 超时后使用当前可用品种继续

2. **合理的等待时间**
   - 品种列表更新通常需要10-30秒
   - 60秒足够覆盖网络不稳定的情况

3. **启动速度**
   - 不影响正常启动速度
   - 后台线程，不阻塞UI

### 为什么每2秒检查一次？

1. **及时响应**
   - 品种列表一旦就绪，尽快开始扫描

2. **资源效率**
   - 不需要过于频繁检查
   - 2秒间隔平衡了响应速度和CPU占用

### 缺失日期检测的4个因素详解

1. **基日过滤**
   - 配置项：`chinastock.base_date`
   - 默认值：2020-01-01
   - 意义：2020年1月1日之前的数据不算缺失
   - 原因：历史数据可能不完整或不需要

2. **上市日期过滤**
   - 数据源：TDX API `get_finance_info`
   - 字段：`ipo_date`
   - 意义：股票上市前的日期不算缺失
   - 示例：2023年上市的股票，2020-2022年没有数据是正常的

3. **交易日历过滤**
   - 数据源：`pandas_market_calendars`
   - 交易所：XSHG（上海证券交易所）
   - 意义：周末、节假日不算缺失
   - 缓存：24小时TTL

4. **最近交易日过滤**
   - 数据源：`TradingCalendar.get_previous_trading_day()`
   - 意义：只检测到最近一个交易日，不检测未来日期
   - 原因：交易日历包含未来日期，但历史数据只到今天
