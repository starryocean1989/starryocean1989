# 1. 导入LoggerMixin
from backend.core.logging_mixin import LoggerMixin

# 2. 继承LoggerMixin
class MyService(BaseService, LoggerMixin):
    def __init__(self):
        super().__init__()
        # 自动获得self.logger
    
    # 3. 使用日志方法
    def my_method(self, data):
        self.log_operation_start("操作名称", param1=value1)
        try:
            result = process(data)
            self.log_operation_success("操作名称", count=len(result))
            return result
        except Exception as e:
            self.log_operation_failure("操作名称", e)
            raise

## 🔍 技术亮点

### 1. 延迟初始化策略
```python
# 数据库已存在时跳过创建
if db_path.exists():
    return

# 只创建关键索引
conn.execute("CREATE INDEX idx_logs_timestamp ON logs(timestamp)")
conn.execute("CREATE INDEX idx_logs_level ON logs(level)")

# 其他索引在后台创建
threading.Thread(target=self._create_remaining_indexes, daemon=True).start()
```

### 2. 错误容错机制
```python
# 数据库错误时静默跳过
if self._db_error:
    return

try:
    # 写入数据库
except Exception as e:
    if not self._db_error:
        print(f"[日志] 数据库写入失败: {e}")
        self._db_error = True
```

### 3. 批量写入优化
```python
# 缓冲区达到阈值时批量写入
with self._batch_lock:
    self._batch_buffer.append(log_data)
    if len(self._batch_buffer) >= 10:
        self._flush_batch_buffer()
```

### 4. 告警抑制策略
