### 模板2：数据下载流程日志

```python
def download_data(self, symbols, start_date, end_date):
    """数据下载示例."""
    self.log_operation_start(
        "数据下载",
        symbol_count=len(symbols),
        date_range=f"{start_date} to {end_date}"
    )
    
    success_count = 0
    failed_count = 0
    
    for i, symbol in enumerate(symbols):
        try:
            # 进度日志（每10%记录一次）
            if i % max(1, len(symbols) // 10) == 0:
                self.logger.info(
                    "下载进度: %d/%d (%.1f%%)",
                    i, len(symbols), (i / len(symbols)) * 100
                )
            
            # 下载单个symbol
            self._download_symbol(symbol, start_date, end_date)
            success_count += 1
            
        except Exception as e:
            self.logger.warning(
                "符号下载失败: symbol=%s, error=%s",
                symbol, str(e)
            )
            failed_count += 1
    
    # 完成日志
    self.log_operation_success(
        "数据下载",
        success=success_count,
        failed=failed_count,
        total=len(symbols)
    )
```

```python
def run_backtest(self, strategy, params):
    """回测执行示例."""
    self.log_operation_start(
        "策略回测",
        strategy=strategy.name,
        params=params
    )
    
    import time
    start_time = time.time()
    
    try:
        # 初始化回测
        self.logger.info("初始化回测环境...")
        self._init_backtest_env(strategy, params)
        
        # 执行回测
        self.logger.info("开始回测执行...")
        result = self._run_backtest_impl(strategy, params)
        
        # 计算性能指标
        duration_ms = (time.time() - start_time) * 1000
        self.log_performance(
            "策略回测",
            duration_ms,
            success=True,
            extra={
                "strategy": strategy.name,
                "total_trades": result.get("total_trades", 0)
            }
        )
        
        self.logger.info(
            "回测完成: return=%.2f%%, sharpe=%.2f, trades=%d",
            result.get("return", 0),
            result.get("sharpe", 0),
            result.get("total_trades", 0)
        )
        
        self.log_operation_success(
            "策略回测",
            return_pct=result.get("return", 0),
            sharpe=result.get("sharpe", 0)
        )
        
        return result
        
    except Exception as e:
        duration_ms = (time.time() - start_time) * 1000
        self.log_performance(
            "策略回测",
            duration_ms,
            success=False
        )
        
        self.log_operation_failure(
            "策略回测",
            e,
            strategy=strategy.name
        )
        raise
```

## 批量写入优化实施

### 当前实现（已有基础）

```python
# backend/core/logging_system.py
class LogManager:
    def __init__(self):
        self._batch_buffer = []
        self._batch_lock = threading.Lock()
        
    def add_log_record(self, log_data):
        with self._batch_lock:
            self._batch_buffer.append(log_data)
            
            # 达到阈值时批量写入
            if len(self._batch_buffer) >= 10:
                self._flush_batch_buffer()
```

## 告警抑制优化实施

### 当前实现（已完成）

```python
# backend/core/alert_system.py
class LogAlertRule:
    def __init__(self, suppression_window=300):
        self.suppression_window = suppression_window
        self._last_trigger_times = {}
    
    def evaluate_log_record(self, log_data):
        # 检查抑制窗口
        suppression_key = f"{self.rule_id}:{log_data.get('module')}:{log_data.get('level')}"
        if suppression_key in self._last_trigger_times:
            last_time = self._last_trigger_times[suppression_key]
            if datetime.now() - last_time < timedelta(seconds=self.suppression_window):
                return False  # 被抑制
        
        # 更新抑制时间
        self._last_trigger_times[suppression_key] = datetime.now()
        return True
```
