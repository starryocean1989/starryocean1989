#### 2. 标准化 - LoggerMixin提供统一日志接口

**创建的工具**:
```python
# backend/core/logging_mixin.py
class LoggerMixin:
    """日志记录器混入类."""
    
    def __init__(self):
        self._logger = logging.getLogger(self.__class__.__name__)
    
    def log_performance(self, operation, duration_ms, success=True, extra=None):
        """记录性能日志."""
        
    def log_operation_start(self, operation, **kwargs):
        """记录操作开始."""
        
    def log_operation_success(self, operation, **kwargs):
        """记录操作成功."""
        
    def log_operation_failure(self, operation, error, **kwargs):
        """记录操作失败."""
```

**使用示例**:
```python
from backend.core.logging_mixin import LoggerMixin

class DataCenterService(BaseService, LoggerMixin):
    def download_data(self, symbols, start_date, end_date):
        self.log_operation_start("数据下载", symbols=len(symbols))
        
        try:
            result = self._do_download(symbols, start_date, end_date)
            self.log_operation_success("数据下载", downloaded=result['count'])
            return result
        except Exception as e:
            self.log_operation_failure("数据下载", e, symbols=symbols)
            raise
```

**效果**:
- 所有服务可轻松集成日志功能
- 日志格式统一、一致
- 自动包含类名、模块名等上下文信息

## 技术亮点

### 1. 延迟初始化策略

```python
def _init_database(self) -> None:
    """初始化数据库（快速模式）."""
    db_path = Path(self.db_path)
    
    # 如果数据库已存在，跳过创建
    if db_path.exists():
        print(f"[启动] 数据库已存在，跳过创建")
        return
    
    # 快速创建基础结构
    with sqlite3.connect(self.db_path, timeout=5.0) as conn:
        # 创建表
        conn.execute("CREATE TABLE IF NOT EXISTS logs (...)")
        
        # 只创建关键索引
        conn.execute("CREATE INDEX IF NOT EXISTS idx_logs_timestamp ON logs(timestamp)")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_logs_level ON logs(level)")
        
        conn.commit()
    
    # 标记需要创建其他索引
    self._needs_index_creation = True

# 首次写入时，在后台创建剩余索引
def add_log_record(self, log_data):
    if self._needs_index_creation:
        threading.Thread(target=self._create_remaining_indexes, daemon=True).start()
```

### 2. 错误容错机制

```python
def add_log_record(self, log_data):
    """添加日志记录（带错误容错）."""
    # 如果数据库有错误，静默跳过
    if self._db_error:
        return
    
    try:
        # 写入数据库
        self.database.add_log_record(log_data)
    except Exception as e:
        # 首次错误时记录，后续静默
        if not self._db_error:
            print(f"[日志] 数据库写入失败: {e}")
            self._db_error = True
```

## 使用指南

### 如何在服务中集成日志

```python
from backend.core.logging_mixin import LoggerMixin
from backend.services.base_and_utils import BaseService

class MyService(BaseService, LoggerMixin):
    def __init__(self):
        super().__init__()
        # LoggerMixin会自动创建self.logger
        
    def _do_initialize(self) -> bool:
        self.logger.info("初始化MyService...")
        try:
            # 初始化逻辑
            self.logger.info("✅ MyService初始化成功")
            return True
        except Exception as e:
            self.logger.error("MyService初始化失败: %s", e, exc_info=True)
            return False
    
    def process_data(self, data):
        self.log_operation_start("数据处理", count=len(data))
        
        import time
        start_time = time.time()
        
        try:
            result = self._do_process(data)
            
            duration_ms = (time.time() - start_time) * 1000
            self.log_performance("数据处理", duration_ms, success=True)
            self.log_operation_success("数据处理", result_count=len(result))
            
            return result
        except Exception as e:
            duration_ms = (time.time() - start_time) * 1000
            self.log_performance("数据处理", duration_ms, success=False)
            self.log_operation_failure("数据处理", e, data_count=len(data))
            raise
```
